;; CxKinship — kinship & age theory: the genl edge and the rules that derive
;; ancestorOf and grandparentOf from parentOf, and olderThan from birthYearOf. Sees the shared
;; upper ontology via CxUniverse; CxWell sees it.

(genlCx CxKinship CxUniverse)
(genlCx CxWell CxKinship)

;; Parenthood is ancestry: parentOf specializes ancestorOf.
(genl parentOf ancestorOf)

(set/forwardRule (implies (and (parentOf ?x ?y) (parentOf ?y ?z)) (grandparentOf ?x ?z)))

(set/backwardRule
  (implies (and (birthYearOf ?x ?bx) (birthYearOf ?y ?by) (lessThan ?bx ?by))
           (olderThan ?x ?y)))

;; Maternity and paternity each entail parenthood, and neither converse holds: a parent
;; is one or the other, and nothing in a parentOf fact says which.
(set/forwardRule (implies (and (fatherOf ?child ?father)) (parentOf ?father ?child)))

(set/forwardRule (implies (and (motherOf ?child ?mother)) (parentOf ?mother ?child)))
