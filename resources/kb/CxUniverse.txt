;; --- situations: change divides them ----------------------------------------
;; A situation either holds unchanged or changes, never both and never neither.  CxAbstract
;; states the two genl edges and the separation; the partition adds the coverage, so a
;; situation denied being an event is concluded a static_situation.
(set/monotonic (partition situation static_situation event))

;; --- the folk taxonomy of organisms ------------------------------------------
;; The kinds of organism as everyday speech sorts them: a species-level kind (cat, oak)
;; under a class-level kind (mammal, tree), and an animal class either with a backbone or
;; without one.  Each level is a disjoint_metatype, so its members are pairwise disjoint
;; without a sentence per pair, and every kind below a member inherits the separation:
;; the vertebrate classes, the invertebrate classes and the plant classes separate the
;; kinds of each, and folk_species separates every two species, so no organism is a cat
;; and a dog, and none is a spider and an owl.

(comment vertebrate "An animal with a backbone: a mammal, a bird, a fish, a reptile or an amphibian.")
(comment invertebrate "An animal without a backbone: an insect, a spider, a snail, a worm.")
;; Every animal has a backbone or lacks one.  The partition installs both genl edges to
;; animal.
(set/monotonic (partition animal vertebrate invertebrate))
(genl amphibian vertebrate)
(genl bird vertebrate)
(genl fish vertebrate)
(genl mammal vertebrate)
(genl reptile vertebrate)
(genl arachnid invertebrate)
(genl insect invertebrate)
;; typeGenl requires its target to be a first-order type: (interArg typeGenl 1 metatype 2 type).
(type vertebrate)
(type invertebrate)
(type plant)

(comment folk_biological_class
         "(folk_biological_class ?type) means that ?type is a class-level kind of organism as everyday speech names it, broader than a species: mammal, insect, tree. The parent of vertebrate_class, invertebrate_class and plant_class, which separating keeps apart, so no class is of two of them.")
(unary_predicate folk_biological_class)
(genlArg folk_biological_class 1 organism)
(metatype folk_biological_class)
;; Every member is a first-order type, whose instances are individual organisms.
(genl folk_biological_class type)
;; The roster installs each part's genl edge to folk_biological_class.
(set/monotonic (separating folk_biological_class vertebrate_class invertebrate_class plant_class))
(typeGenl vertebrate_class vertebrate)

(comment invertebrate_class
         "(invertebrate_class ?type) means that ?type is one of the classes of animal without a backbone. A disjoint_metatype, so its members are pairwise disjoint and every kind below one is separated from every kind below another.")
(unary_predicate invertebrate_class)
(genlArg invertebrate_class 1 invertebrate)
(metatype invertebrate_class)
(set/monotonic (disjoint_metatype invertebrate_class))
(invertebrate_class arachnid)
(invertebrate_class insect)
(typeGenl invertebrate_class invertebrate)

(comment plant_class
         "(plant_class ?type) means that ?type is one of the folk classes of plant, such as tree, flower and grass. A disjoint_metatype, so its members are pairwise disjoint: an oak is never a rose, and grass is neither. Not a partition of plant, since a fern or a moss is none of the three.")
(unary_predicate plant_class)
(genlArg plant_class 1 plant)
(metatype plant_class)
(set/monotonic (disjoint_metatype plant_class))
(plant_class flower)
(plant_class grass)
(plant_class tree)
(typeGenl plant_class plant)

(comment folk_species
         "(folk_species ?type) means that ?type is a species-level kind of organism as everyday speech names it: cat, dog, oak, human. A disjoint_metatype, so no organism is of two species, and never a class-level kind: a species is not a folk_biological_class.")
(unary_predicate folk_species)
(genlArg folk_species 1 organism)
(metatype folk_species)
;; Every member is a first-order type, whose instances are individual organisms.
(genl folk_species type)
(set/monotonic (disjoint_metatype folk_species))
(set/monotonic (disjoint folk_biological_class folk_species))
